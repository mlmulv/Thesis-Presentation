\begin{frame}
 \frametitle{Computational Approximation of the ICING model} 
 \begin{itemize}
    \item The ICING model equations are defined in continuous-time
    \item Discretization of the model is required for the presented estimation methods
 \end{itemize}
  \begin{exampleblock}{For an example dynamic $F$}
   \begin{align*}
    t_k &= t_{k-1} + \Delta, \\
    \dot{F} &\approx \frac{F(t_{k}) - F(t_{k} - \Delta)}{\Delta}, \\
    F_{k} &\approx F_{k-1} + \Delta \dot{F}.
   \end{align*} 
  \end{exampleblock}
  \note[item]{According to Euler's method}
  \note[item]{Discretization error is bounded by $\Delta$}
\end{frame}

\begin{frame}
  \frametitle{Bayesian Filtering}
  The discrete-time dynamic ICING model can be represented in the stochastic state-space. \\ 
  \begin{columns}
    \begin{column}{0.45\linewidth} 
      \begin{block}{State-space Representation}
        \begin{align*}
          x_{k} &\sim p(x_{k} \mid x_{k-1},\,u_{k-1}) \\
          y_{k} &\sim p(y_{k} \mid x_{k})
        \end{align*}
      \end{block} 
    \end{column}
    \begin{column}{0.48\linewidth}
    \begin{block}{Parameter Description}
      \begin{table}
        \centering
        \begin{tabularx}{0.96\linewidth}{>{\raggedright\arraybackslash}X r}
          \toprule
          \textbf{System Variable} & \textbf{Notation} \\
          \midrule
          Dynamics & $x_k$ \\
          Inputs & $u_k$ \\
          Measurements & $y_k$ \\
          \bottomrule
        \end{tabularx}
      \end{table} 
    \end{block}
    \end{column}
  \end{columns}
  \vspace{1em}
  There are various methods of optimal filtering, and each has their own assumptions and limitations
  \note[item]{The measurement often depends on the hospital, but will be discussed more later \\}
  \note[item]{Just the EKF and the PF will be in the topic of this thesis}
\end{frame}

\begin{frame}
  \frametitle{Extended Kalman Filter}
  Approximates \alert{nonlinear} and \alert{Gaussian} dynamic and measurement models as a linear transformation.
  
  \begin{block}{}
    \begin{enumerate}
    \item \textit{Prediction step} 
    \begin{equation} \label{eq:Predict}
        \begin{aligned}
            m_{k}^- &= f(m_{k-1},\,u_{k-1}), \\
            P_{k}^- &= F_x(m_{k-1})P_{k-1}F_x^{\sf T}(m_{k-1}) + F_q(m_{k-1})Q_{k-1}F_q^{\sf T}(m_{k-1}). \notag
        \end{aligned}
    \end{equation}
    \item \textit{Update step} 
    \begin{equation} \label{eq:Update}
        \begin{aligned}
            v_{k} &= y_{k} - h(m_k^-), \\
            S_{k} &= H_x(m_k^-)P_{k}^-H_x^{\sf T}(m_k^-) + H_r(m_k^-)R_kH_r^{\sf T}(m_k^-), \\
            K_k &= P_k^-H_k^{\sf T}(m_k^-)S_k^{-1}, \\
            m_k &= m_k^- + K_kv_k, \\
            P_k &= P_k^- - K_kS_kK_k^{\sf T}.
        \end{aligned} \notag
    \end{equation}
    \end{enumerate} 
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Bootstrap Particle Filter}
  Formulates the posterior of the dynamic function as a Monte Carlo approximation.
  \begin{block}{}
    \begin{enumerate}
    \item Sample $N$ particles from the dynamic function
    \begin{align}
        x_k^{(i)} \sim p(x_k \mid x_{k-1}^{(i)},\,u_{k-1}) \quad i = 1,\dots, N. \notag
    \end{align}
    \item Update weights from the measurement
    \begin{equation}
        \begin{aligned}
            w_k^{*(i)} &\propto p(y_k \mid x_k^{(i)})  \quad i = 1,\dots, N  ,\\
            w_k^{(i)} &= \frac{w_k^{*(i)}}{\sum_{j=1}^{N} w_k^{*(j)}} \quad i = 1,\dots, N .
        \end{aligned} \notag
    \end{equation}
    \item Systematically resample particles from a distribution proportional to the weights
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Estimation of Patient Insulin-Glucose Dynamics}
  \begin{itemize}
    \item Previous studies have employed Bayesian filtering methods:
      \begin{itemize}
        \item Extended and unscented Kalman filters
        \item Particle filters
      \end{itemize}
    \item Practical implementation is often insufficiently addressed:
      \begin{itemize}
        \item Constant nutrition and insulin inputs
        \item Known initial insulin–glucose dynamics
    \end{itemize}
    \item Estimator configurations and performance are not systematically compared.
  \end{itemize}
