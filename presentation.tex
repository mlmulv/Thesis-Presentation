\begin{frame}
  \frametitle{Stress Hyperglycaemia}
  \begin{enumerate}
    \item<1-> Patients suffering from critical illnesses incur metabolic changes from acute stress
    \note<1->[item] {Inflammatory response secretes stress hormones}
    \note<1->[item] {Acute myocardial infarction, sepsis, stroke}
    \item<2-> Instances of stress-induced hyperglycaemia have been identified to accompany the onset of \alert{mortality} 
    \item<3-> In a controlled study, tightly maintaining glycaemic levels (4 mmol/L - 6 mmol/L) reduced mortalities by 50\% compared to conventional methods
  \end{enumerate}
\end{frame}
