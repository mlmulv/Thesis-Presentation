\begin{frame}
  \frametitle{Stress Hyperglycaemia in ICU Patients}
  \begin{enumerate}
    \item<1-> Patients suffering from critical illnesses are subjected to metabolic changes from acute stress
    \note<1->[item] {Inflammatory response secretes stress hormones}
    \note<1->[item] {Acute myocardial infarction, sepsis, stroke}
    \item<2-> Instances of stress-induced hyperglycaemia have been identified to accompany the onset of \alert{mortality} 
    \item<3-> Tightly maintaining glycaemic levels in ranges 4 mmol/L to 6 mmol/L reduced mortalities by 50\% \parencite{berghe_intensive_2001}
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Limitations of Intensive Insulin Therapy}
  \begin{columns}
    \begin{column}{5cm}
      \begin{enumerate}
        \item<1-> Solely utilizing insulin infusions to tightly control blood glucose struggles to address patient variability
        \note[item]<1->{Infrequent measurements, not considering nutrition, and not complex to monitor dynamics changes in patient health}
        \item<2-> Poor patient adaptability additionally results in cases of hypoglycaemia and high glycaemic variation
      \end{enumerate}
    \end{column}
    \begin{column}{5cm}    
      \begin{figure}
